==== ./spring_grpc/src/main/proto\channel.proto ====
syntax = "proto3";
package channel;
option java_package = "com.drocsid.grpc.proto";
option java_multiple_files = true;
import "common.proto";
import "role.proto";
import "message.proto";


service ChannelService {

  rpc GetChannel(ChannelInfoReq) returns (Channel);
  rpc UpdateChannel(UpdateChannelRequest) returns (Channel);
  rpc DeleteChannel(ChannelInfoReq) returns (common.ResponseMessage);

  // messages
  rpc GetMessages(MessageBucketReq) returns (dmessage.MessageList);
  rpc CreateMessage(CreateMessageReq) returns (dmessage.Message);
  rpc DeleteMessage(MessageReq) returns (common.ResponseMessage);
}

message Channel {
  string channel_id = 1;
  string name = 2;
  string guild_id = 3;
  role.RoleList overrides = 4;
}

message CreateChannelRequest {
  string caller_id = 1;
  string name = 2;
  string guild_id = 3;
}

message ChannelInfoReq {
  string caller_id = 1;
  string channel_id = 2;
}

message UpdateChannelRequest {
  string caller_id = 1;
  Channel channel = 2;
}

message ChannelList {
  repeated Channel channels = 1;
}

message MessageBucketReq {
  ChannelInfoReq req = 1;
  int32 offset = 2;
  int32 count = 3;
}

message CreateMessageReq {
  ChannelInfoReq req = 1;
  dmessage.GeneralCreateMessageRequest message = 2;
}

message MessageReq {
  ChannelInfoReq req = 1;
  dmessage.GeneralMessageReq message_req = 2;
}

syntax = "proto3";
package guild;
option java_package = "com.drocsid.grpc.proto";
option java_multiple_files = true;
import "common.proto";
import "role.proto";
import "guild_user.proto";
import "user.proto";
import "channel.proto";


service GuildService {
  // basic managment
  rpc CreateGuild(CreateGuildRequest) returns (Guild);
  rpc GetGuild(GuildInfoReq) returns (Guild);
  rpc UpdateGuild(UpdateGuildRequest) returns (Guild);
  rpc DeleteGuild(GuildInfoReq) returns (common.ResponseMessage);
  rpc GetAllGuilds(user.UserId) returns (GuildList);
  rpc GetAllChannels(GuildInfoReq) returns (channel.ChannelList);

  // role managment
  rpc GetRole(role.RoleReq) returns (role.Role);
  rpc CreateRole(role.CreateRoleReq) returns (role.Role);
  rpc GetRoles(GuildInfoReq) returns (role.RoleList);
  rpc UpdateRole(role.UpdateRoleReq) returns (role.Role);

  // channel creation
  rpc CreateChannel(channel.CreateChannelRequest) returns (channel.Channel);

  // guild user managment
  rpc AddUser(guild_user.AddUser) returns (guild_user.GuildUser);
  rpc UpdateUser(guild_user.UpdateUserReq) returns (guild_user.GuildUser); // future
  rpc GetUser(guild_user.GuildUserReq) returns (guild_user.GuildUser);
  rpc GetUsers(GuildInfoReq) returns (guild_user.GuildUserList);
  rpc DeleteUser(guild_user.GuildUserReq) returns (common.ResponseMessage);
}

message Guild {
  string guild_id = 1;
  string name = 2;
  string icon = 3; // not used, future
  string owner_id = 4;
  repeated role.Role roles = 5;
}

message CreateGuildRequest {
  string caller_id = 1;
  string name = 2;
  string icon = 3; // not used, future
}

message GuildInfoReq {
  string guild_id = 1;
  string caller_id = 2;
}

message UpdateGuildRequest {
  string caller_id = 1;
  Guild guild = 2;
}

message GuildList {
  repeated Guild guilds = 1;
}
